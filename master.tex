%-------------------------
% Custom commands
\newcommand{\resumeItem}[1]{
  \item\small{
    {#1}
  }
}

% A bullet that opens with a bolded lead-in, so the eye can catch what the
% bullet is about before reading it. Wraps \resumeItem rather than repeating
% its body, so the two never drift apart.
\newcommand{\resumeItemLead}[2]{\resumeItem{\textbf{#1} --- #2}}

% One-sentence pitch under the Summary heading. \small matches the body -- the
% document default is 11pt, which would render larger than every bullet below.
\newcommand{\resumeSummary}[1]{{\small #1}\par\vspace{2pt}}

\newcommand{\resumeSubheading}[4]{
  \vspace{7pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-2pt}
}

\newcommand{\resumeSubSubheading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeProjectHeading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \small#1 & #2 \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeSubItem}[1]{\resumeItem{#1}\vspace{-4pt}}

\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}

\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0.15in, label={}]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
% Spacing knobs. These values are the slack the one-page layout runs on: raise
% them for more air, lower them to absorb added content. Bumping the
% section/subheading gaps by ~4pt each is roughly what tips this onto a second
% page, so change them one step at a time and check `pdfinfo pdf/*.pdf`.
%
% leftmargin on the item list is a width knob, not a vertical one. Bullet bodies
% are written to fit on a single line at this indent, so widening it wraps them.
\newcommand{\resumeItemListStart}{\begin{itemize}[leftmargin=0.15in, itemsep=4pt, topsep=2pt, parsep=0pt]}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{3pt}}

% Education reuses the two-column subheading layout, but takes its arguments
% in school/location/degree/dates order.
\newcommand{\resumeEducation}[4]{\resumeSubheading{#1}{#2}{#3}{#4}}

%-------------------------
% Skills list
%
% Every skill line sits inside ONE \item separated by \\, which keeps the
% block tightly packed. The flag emits the separator BEFORE each entry except
% the first, so the list never ends on a trailing \\ (which would add a stray
% blank line).
\newif\ifresumefirstskill
\newcommand{\resumeSkillListStart}{%
  \begin{itemize}[leftmargin=0.15in, label={}]\small\item\resumefirstskilltrue
}
\newcommand{\resumeSkill}[2]{%
  \ifresumefirstskill\resumefirstskillfalse\else\\\fi\textbf{#1}{: #2}%
}
\newcommand{\resumeSkillListEnd}{\end{itemize}}

% Contact header -- the only thing that differs between print and online.
\ifx\format\resumeformatprint
  % PRINT: icon glyphs. Nothing is clickable on paper, so labels are dropped
  % in favour of logos. The trailing "\ " after each icon is a real space.
  \newcommand{\resumeLink}[2]{#2}
  \newcommand{\resumeHeader}[5]{
    \begin{center}
      \textbf{\Huge \scshape #1} \\ \vspace{6pt}
      \small \faPhone\ #2 $|$ \faEnvelope\ #3 \\ \vspace{4pt}
      \small \faLinkedin\ #4 $|$ \faGithub\ #5
    \end{center}
  }
\else
  % ONLINE: underlined text hyperlinks. Kept text-only on purpose -- ATS
  % parsers frequently mangle or drop icon-font glyphs along with the
  % contact detail beside them.
  \newcommand{\resumeLink}[2]{\href{#1}{\underline{#2}}}
  \newcommand{\resumeHeader}[5]{
    \begin{center}
      \textbf{\Huge \scshape #1} \\ \vspace{6pt}
      \small #2 $|$ #3 \\ \vspace{4pt}
      \small #4 $|$ #5
    \end{center}
  }
\fi
